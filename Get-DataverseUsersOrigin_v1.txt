<#
.SYNOPSIS
Lists Dataverse security roles and categorises each role as Microsoft, Custom, or Managed non-Microsoft.

.DESCRIPTION
The script calls the Dataverse Web API and uses role metadata to infer origin:
- roletemplateid: links roles created from Microsoft role templates.
- ismanaged: tells whether the role is part of a managed solution.
- solution publisher: distinguishes Microsoft managed content from ISV/customer managed content.
- product solution/name patterns: identifies common predefined Dynamics 365 product roles and reports ProductName.
- product role purpose patterns: labels common Dynamics 365 product roles as Functional, App Access, Internal, or Administrative.
- role documentation lookup: adds a short Microsoft-documented description, AccessGranted text, and source link for known predefined roles.

Because copied or solution-imported roles can be ambiguous, the script also emits an Evidence column.

.PARAMETER EnvironmentUrl
Dataverse environment URL, for example https://contoso.crm.dynamics.com.

.PARAMETER AccessToken
Optional OAuth bearer token for the Dataverse environment. If omitted, the script uses Az.Accounts
and prompts through Connect-AzAccount when needed.

.PARAMETER OutputPath
Optional CSV output path. Defaults to .\DataverseSecurityRoles-Origin.csv.

.PARAMETER IncludeBusinessUnitCopies
By default, the script returns one row per root role. Use this switch to list every business-unit copy.

.EXAMPLE
.\Get-DataverseSecurityRoleOrigin.ps1 -EnvironmentUrl "https://contoso.crm.dynamics.com"

.EXAMPLE
$token = (Get-AzAccessToken -ResourceUrl "https://contoso.crm.dynamics.com").Token
.\Get-DataverseSecurityRoleOrigin.ps1 -EnvironmentUrl "https://contoso.crm.dynamics.com" -AccessToken $token
#>

function Get-DataverseAccessToken {
    param([string]$ResourceUrl)

    if ($AccessToken) {
        return $AccessToken
    }

    if (-not (Get-Module -ListAvailable -Name Az.Accounts)) {
        throw "Az.Accounts is required when -AccessToken is not supplied. Install it with: Install-Module Az.Accounts -Scope CurrentUser"
    }

    Import-Module Az.Accounts -ErrorAction Stop

    try {
        $context = Get-AzContext -ErrorAction Stop
    }
    catch {
        $context = $null
    }

    if (-not $context) {
        Connect-AzAccount | Out-Null
    }

    $tokenResponse = Get-AzAccessToken -ResourceUrl $ResourceUrl
    if ($tokenResponse.Token -is [System.Security.SecureString]) {
        $credential = [System.Net.NetworkCredential]::new("", $tokenResponse.Token)
        return $credential.Password
    }

    return [string]$tokenResponse.Token
}

function Invoke-DataversePagedGet {
    param(
        [string]$Url,
        [hashtable]$Headers
    )

    $items = New-Object System.Collections.Generic.List[object]
    $nextUrl = $Url

    while ($nextUrl) {
        $response = Invoke-RestMethod -Method Get -Uri $nextUrl -Headers $Headers
        foreach ($item in $response.value) {
            $items.Add($item)
        }

        $nextProperty = "@odata.nextLink"
        if ($response.PSObject.Properties.Name -contains $nextProperty) {
            $nextUrl = $response.$nextProperty
        }
        else {
            $nextUrl = $null
        }
    }

    return $items
}

function Test-MicrosoftPublisher {
    param(
        [string]$PublisherName,
        [string]$PublisherUniqueName,
        [string]$CustomizationPrefix
    )

    $values = @($PublisherName, $PublisherUniqueName, $CustomizationPrefix) | Where-Object { -not [string]::IsNullOrWhiteSpace($_) }
    foreach ($value in $values) {
        if ($value -match '^(Microsoft|Microsoft Corporation|MicrosoftCorporation|Microsoft First Party)$') {
            return $true
        }
        if ($value -match '^msdyn' -or $value -match '^msfp' -or $value -eq "microsoft") {
            return $true
        }
    }

    return $false
}

function Get-ProductRolePurpose {
    param(
        [string]$RoleName,
        [string]$ProductName,
        [bool]$IsKnownPredefinedRole
    )

    if ([string]::IsNullOrWhiteSpace($ProductName)) {
        return $null
    }

    if ($IsKnownPredefinedRole) {
        if ($RoleName -match '(?i)administrator|admin|customizer|maker|owner') {
            return "Administrative"
        }

        return "Functional"
    }

    $internalRoleNames = @(
        "Forecast Appuser",
        "Dynamics 365 Marketing Application User",
        "EventManagement S2S Inbound",
        "Marketing, Business App Access",
        "Marketing Services User",
        "Marketing Services Application User",
        "IoT - Endpoint User",
        "CFS - IoT Alert Process Flow"
    )

    if ($internalRoleNames -contains $RoleName) {
        return "Internal"
    }

    if ($RoleName -match '(?i)app access|appaccess|app profile user|sales team member|knowledge search app user') {
        return "App Access"
    }

    if ($RoleName -match '(?i)administrator|admin|customizer|owner') {
        return "Administrative"
    }

    return "Functional"
}

function Get-RoleOrigin {
    param(
        [bool]$IsKnownPredefinedRole,
        [bool]$IsKnownD365ProductRole,
        [string]$ProductName,
        [bool]$HasRoleTemplate,
        [bool]$IsManaged,
        [bool]$IsMicrosoftPublisher,
        [string]$PublisherName
    )

    if ($IsKnownD365ProductRole -and ($IsManaged -or $IsMicrosoftPublisher)) {
        return @{
            Category = "Microsoft Managed"
            Evidence = "Role matches a predefined Dynamics 365 product role or Microsoft product solution. Product: $ProductName."
        }
    }

    if ($IsKnownPredefinedRole) {
        return @{
            Category = "Microsoft Managed"
            Evidence = "Role name matches the Microsoft predefined Dataverse security role list."
        }
    }

    if ($HasRoleTemplate -and $IsMicrosoftPublisher) {
        return @{
            Category = "Microsoft Managed"
            Evidence = "Has roletemplateid and belongs to a Microsoft-published solution."
        }
    }

    if ($HasRoleTemplate -and -not $PublisherName) {
        return @{
            Category = "Likely Microsoft Managed"
            Evidence = "Has roletemplateid, but the solution publisher could not be resolved."
        }
    }

    if ($HasRoleTemplate) {
        return @{
            Category = "Custom or copied from Microsoft template"
            Evidence = "Has roletemplateid, but publisher is not Microsoft."
        }
    }

    if ($IsManaged -and $IsMicrosoftPublisher) {
        return @{
            Category = "Microsoft Managed"
            Evidence = "Managed role in a Microsoft-published solution, but no roletemplateid."
        }
    }

    if ($IsManaged) {
        return @{
            Category = "Managed non-Microsoft"
            Evidence = "Managed role from a non-Microsoft publisher."
        }
    }

    return @{
        Category = "Custom"
        Evidence = "No roletemplateid and role is unmanaged."
    }
}

function Get-PrivilegeDepthRank {
    param([int]$PrivilegeDepthMask)

    if (($PrivilegeDepthMask -band 8) -eq 8) {
        return 4
    }
    if (($PrivilegeDepthMask -band 4) -eq 4) {
        return 3
    }
    if (($PrivilegeDepthMask -band 2) -eq 2) {
        return 2
    }
    if (($PrivilegeDepthMask -band 1) -eq 1) {
        return 1
    }

    return 0
}

function Get-PrivilegeDepthLabel {
    param([int]$PrivilegeDepthMask)

    switch (Get-PrivilegeDepthRank -PrivilegeDepthMask $PrivilegeDepthMask) {
        4 { return "Global" }
        3 { return "Deep" }
        2 { return "Local" }
        1 { return "Basic" }
        default { return "None" }
    }
}

function Get-RolePrivilegeProperty {
    param(
        [object]$RolePrivilege,
        [string]$Name
    )

    $value = Get-PropertyValue -Object $RolePrivilege -Name $Name
    if ($value) {
        return $value
    }

    return Get-PropertyValue -Object $RolePrivilege -Name "_$($Name)_value"
}

function Get-HighRiskPrivilegeCategory {
    param(
        [string]$PrivilegeName,
        [int]$DepthRank
    )

    if ([string]::IsNullOrWhiteSpace($PrivilegeName)) {
        return $null
    }

    # Validate depth per-pattern: some categories only count at higher privilege depth
    $sensitivePrivilegePatterns = @(
        @{ Pattern = '^prv(Create|Write|Delete|Assign|Append|AppendTo).*Organization$'; Category = "Organization administration"; MinDepthRank = 0 },
        @{ Pattern = '^prv(Create|Write|Delete|Append|AppendTo).*Publisher$'; Category = "Publisher administration"; MinDepthRank = 0 },
        @{ Pattern = '^prv(Create|Write|Delete|Append|AppendTo).*Solution$|^prvImportSolution$|^prvExportSolution$'; Category = "Solution administration"; MinDepthRank = 0 },
        @{ Pattern = '^prvPublishCustomization$|^prv(Create|Write|Delete).*Customization$'; Category = "Customization publishing"; MinDepthRank = 0 },
        @{ Pattern = '^prv(Create|Write|Delete).*Entity$|^prv(Create|Write|Delete).*Attribute$|^prv(Create|Write|Delete).*Relationship$'; Category = "Metadata customization"; MinDepthRank = 0 },
        @{ Pattern = '^prv(Create|Write|Delete|Append|AppendTo).*Plugin|^prv(Create|Write|Delete|Append|AppendTo).*SdkMessageProcessingStep'; Category = "Plugin/extension administration"; MinDepthRank = 0 },
        @{ Pattern = '^prv(Create|Write|Delete|Assign|Append|AppendTo).*Role$|^prvAssignRole$|^prvAssignUserRoles$'; Category = "Security role administration"; MinDepthRank = 0 },
        @{ Pattern = '^prv(Create|Write|Delete|Assign|Append|AppendTo).*Privilege$|^prvReadPrivilege$'; Category = "Privilege administration"; MinDepthRank = 4 },
        @{ Pattern = '^prv(Create|Write|Delete|Assign|Append|AppendTo).*SystemUser$|^prv(Create|Write|Delete|Assign).*User$'; Category = "User administration"; MinDepthRank = 0 },
        @{ Pattern = '^prv(Create|Write|Delete|Assign|Append|AppendTo).*Team$'; Category = "Team administration"; MinDepthRank = 3 },
        @{ Pattern = '^prv(Create|Write|Delete|Assign|Append|AppendTo).*BusinessUnit$'; Category = "Business unit administration"; MinDepthRank = 0 },
        @{ Pattern = '^prv(Create|Write|Delete|Append|AppendTo).*EnvironmentVariable|^prv(Create|Write|Delete|Append|AppendTo).*ConnectionReference'; Category = "Application configuration administration"; MinDepthRank = 2 },
        @{ Pattern = '^prv(Create|Write|Delete|Activate|Append|AppendTo).*Workflow$|^prv(Create|Write|Delete|Activate).*Process'; Category = "Process/workflow administration"; MinDepthRank = 3 }
    )

    foreach ($pattern in $sensitivePrivilegePatterns) {
        if ($PrivilegeName -match $pattern.Pattern) {
            $min = if ($pattern.ContainsKey('MinDepthRank')) { $pattern.MinDepthRank } else { 1 }
            if ($DepthRank -ge $min) {
                return $pattern.Category
            }
            else {
                return $null
            }
        }
    }

    return $null
}

function Get-PrivilegeSimilarityPercent {
    param(
        [hashtable]$RolePrivilegeMap,
        [hashtable]$BaselinePrivilegeMap
    )

    if (-not $BaselinePrivilegeMap -or $BaselinePrivilegeMap.Count -eq 0) {
        return $null
    }

    $matchCount = 0
    foreach ($baselinePrivilege in $BaselinePrivilegeMap.GetEnumerator()) {
        if ($RolePrivilegeMap.ContainsKey($baselinePrivilege.Key) -and $RolePrivilegeMap[$baselinePrivilege.Key].DepthRank -ge $baselinePrivilege.Value.DepthRank) {
            $matchCount++
        }
    }

    return [math]::Round(($matchCount / $BaselinePrivilegeMap.Count) * 100, 1)
}

function Get-AdminRiskProfile {
    param(
        [hashtable]$RolePrivilegeMap,
        [hashtable]$SystemAdminPrivilegeMap,
        [hashtable]$SystemCustomizerPrivilegeMap
    )

    if (-not $RolePrivilegeMap) {
        $RolePrivilegeMap = @{}
    }

    $systemAdminSimilarity = Get-PrivilegeSimilarityPercent -RolePrivilegeMap $RolePrivilegeMap -BaselinePrivilegeMap $SystemAdminPrivilegeMap
    $systemCustomizerSimilarity = Get-PrivilegeSimilarityPercent -RolePrivilegeMap $RolePrivilegeMap -BaselinePrivilegeMap $SystemCustomizerPrivilegeMap

    $highRiskPrivileges = @($RolePrivilegeMap.Values | Where-Object { $_.IsHighRisk })
    $globalHighRiskPrivileges = @($highRiskPrivileges | Where-Object { $_.DepthRank -ge 4 })
    $criticalGlobalPrivileges = @($globalHighRiskPrivileges | Where-Object {
            $_.HighRiskCategory -in @(
                "Security role administration",
                "Privilege administration",
                "User administration",
                "Business unit administration",
                "Organization administration",
                "Plugin/extension administration"
            )
        })

    $riskLevel = "Normal"
    $evidence = New-Object System.Collections.Generic.List[string]

    if ($null -ne $systemAdminSimilarity -and $systemAdminSimilarity -ge 80) {
        $riskLevel = "Critical"
        $evidence.Add(">=80% privilege-depth match to System Administrator")
    }
    elseif ($criticalGlobalPrivileges.Count -gt 0) {
        $riskLevel = "Critical"
        $evidence.Add("Has global critical admin privileges")
    }
    elseif (($null -ne $systemAdminSimilarity -and $systemAdminSimilarity -ge 60) -or ($null -ne $systemCustomizerSimilarity -and $systemCustomizerSimilarity -ge 70) -or $globalHighRiskPrivileges.Count -ge 5) {
        $riskLevel = "High"
        $evidence.Add("High admin/customizer similarity or multiple global high-risk privileges")
    }
    elseif ($globalHighRiskPrivileges.Count -gt 0 -or $highRiskPrivileges.Count -ge 5 -or ($null -ne $systemCustomizerSimilarity -and $systemCustomizerSimilarity -ge 50)) {
        $riskLevel = "Medium"
        $evidence.Add("Contains global high-risk privileges, multiple high-risk privileges, or customizer-like access")
    }
    elseif ($highRiskPrivileges.Count -gt 0) {
        $riskLevel = "Low"
        $evidence.Add("Contains limited high-risk privileges")
    }

    if ($globalHighRiskPrivileges.Count -gt 0) {
        $evidence.Add("$($globalHighRiskPrivileges.Count) high-risk privileges at Global depth")
    }

    $highRiskNames = @($highRiskPrivileges |
        Sort-Object HighRiskCategory, Name |
        Select-Object -First 50 |
        ForEach-Object { "$($_.Name)[$($_.DepthLabel)]" })

    if ($highRiskPrivileges.Count -gt 50) {
        $highRiskNames += "...+$($highRiskPrivileges.Count - 50) more"
    }

    [pscustomobject]@{
        AdminRiskLevel                    = $riskLevel
        SystemAdminSimilarityPercent      = $systemAdminSimilarity
        SystemCustomizerSimilarityPercent = $systemCustomizerSimilarity
        HighRiskPrivilegeCount            = $highRiskPrivileges.Count
        GlobalHighRiskPrivilegeCount      = $globalHighRiskPrivileges.Count
        CriticalGlobalPrivilegeCount      = $criticalGlobalPrivileges.Count
        HighRiskPrivilegeNames            = ($highRiskNames -join "; ")
        AdminRiskEvidence                 = ($evidence -join "; ")
    }
}

$baseUrl = Normalize-EnvironmentUrl -Url $EnvironmentUrl
$token = Get-DataverseAccessToken -ResourceUrl $baseUrl

$headers = @{
    Authorization      = "Bearer $token"
    Accept             = "application/json"
    "OData-MaxVersion" = "4.0"
    "OData-Version"    = "4.0"
    Prefer             = 'odata.include-annotations="OData.Community.Display.V1.FormattedValue"'
}

$apiRoot = "$baseUrl/api/data/v9.2"

Write-Host "Reading role templates..."
$roleTemplateUrl = "$apiRoot/roletemplates?`$select=roletemplateid,name"
$roleTemplates = Invoke-DataversePagedGet -Url $roleTemplateUrl -Headers $headers
$roleTemplateById = @{}
foreach ($template in $roleTemplates) {
    $roleTemplateById[[string]$template.roletemplateid] = $template.name
}

Write-Host "Reading solutions and publishers..."
$solutionUrl = "$apiRoot/solutions?`$select=solutionid,uniquename,friendlyname,ismanaged&`$expand=publisherid(`$select=publisherid,uniquename,friendlyname,customizationprefix)"
$solutions = Invoke-DataversePagedGet -Url $solutionUrl -Headers $headers
$solutionById = @{}
foreach ($solution in $solutions) {
    $solutionById[[string]$solution.solutionid] = $solution
}

Write-Host "Reading security roles..."
$roleSelect = "roleid,roleidunique,name,description,ismanaged,solutionid,createdon,modifiedon,_createdby_value,_modifiedby_value,_roletemplateid_value,_parentrootroleid_value,_businessunitid_value"
$roleUrl = "$apiRoot/roles?`$select=$roleSelect&`$orderby=name asc"
$roles = Invoke-DataversePagedGet -Url $roleUrl -Headers $headers

if (-not $IncludeBusinessUnitCopies) {
    $rootOrFirstRoleByKey = @{}
    foreach ($role in $roles) {
        $parentRootRoleId = [string](Get-PropertyValue -Object $role -Name "_parentrootroleid_value")
        $roleId = [string](Get-PropertyValue -Object $role -Name "roleid")
        $key = if ($parentRootRoleId) { $parentRootRoleId } else { $roleId }

        if (-not $rootOrFirstRoleByKey.ContainsKey($key)) {
            $rootOrFirstRoleByKey[$key] = $role
        }
        elseif ($roleId -eq $key) {
            $rootOrFirstRoleByKey[$key] = $role
        }
    }

    $roles = $rootOrFirstRoleByKey.Values
}

Write-Host "Reading privileges..."
$privilegeUrl = "$apiRoot/privileges?`$select=privilegeid,name"
$privileges = Invoke-DataversePagedGet -Url $privilegeUrl -Headers $headers
$privilegeNameById = @{}
foreach ($privilege in $privileges) {
    $privilegeId = [string](Get-PropertyValue -Object $privilege -Name "privilegeid")
    if (-not [string]::IsNullOrWhiteSpace($privilegeId)) {
        $privilegeNameById[$privilegeId.ToLowerInvariant()] = Get-PropertyValue -Object $privilege -Name "name"
    }
}

Write-Host "Reading role privileges..."
$rolePrivilegeUrl = "$apiRoot/roleprivilegescollection?`$select=roleid,privilegeid,privilegedepthmask"
$rolePrivileges = Invoke-DataversePagedGet -Url $rolePrivilegeUrl -Headers $headers

$selectedRoleIds = @{}
foreach ($role in $roles) {
    $roleId = [string](Get-PropertyValue -Object $role -Name "roleid")
    if (-not [string]::IsNullOrWhiteSpace($roleId)) {
        $selectedRoleIds[$roleId.ToLowerInvariant()] = $true
    }
}

$rolePrivilegeMapByRoleId = @{}
foreach ($rolePrivilege in $rolePrivileges) {
    $rolePrivilegeRoleId = [string](Get-RolePrivilegeProperty -RolePrivilege $rolePrivilege -Name "roleid")
    $rolePrivilegePrivilegeId = [string](Get-RolePrivilegeProperty -RolePrivilege $rolePrivilege -Name "privilegeid")

    if ([string]::IsNullOrWhiteSpace($rolePrivilegeRoleId) -or [string]::IsNullOrWhiteSpace($rolePrivilegePrivilegeId)) {
        continue
    }

    $rolePrivilegeRoleId = $rolePrivilegeRoleId.ToLowerInvariant()
    $rolePrivilegePrivilegeId = $rolePrivilegePrivilegeId.ToLowerInvariant()

    if (-not $selectedRoleIds.ContainsKey($rolePrivilegeRoleId)) {
        continue
    }

    if (-not $privilegeNameById.ContainsKey($rolePrivilegePrivilegeId)) {
        continue
    }

    $privilegeName = [string]$privilegeNameById[$rolePrivilegePrivilegeId]
    if ([string]::IsNullOrWhiteSpace($privilegeName)) {
        continue
    }

    $depthMask = [int](Get-PropertyValue -Object $rolePrivilege -Name "privilegedepthmask")
    $depthRank = Get-PrivilegeDepthRank -PrivilegeDepthMask $depthMask
    $highRiskCategory = Get-HighRiskPrivilegeCategory -PrivilegeName $privilegeName -DepthRank $depthRank
    $privilegeSummary = [pscustomobject]@{
        Name             = $privilegeName
        DepthMask        = $depthMask
        DepthRank        = $depthRank
        DepthLabel       = Get-PrivilegeDepthLabel -PrivilegeDepthMask $depthMask
        IsHighRisk       = -not [string]::IsNullOrWhiteSpace($highRiskCategory)
        HighRiskCategory = $highRiskCategory
    }

    if (-not $rolePrivilegeMapByRoleId.ContainsKey($rolePrivilegeRoleId)) {
        $rolePrivilegeMapByRoleId[$rolePrivilegeRoleId] = @{}
    }

    $rolePrivilegeMap = $rolePrivilegeMapByRoleId[$rolePrivilegeRoleId]
    if (-not $rolePrivilegeMap.ContainsKey($privilegeName) -or $depthRank -gt $rolePrivilegeMap[$privilegeName].DepthRank) {
        $rolePrivilegeMap[$privilegeName] = $privilegeSummary
    }
}

$systemAdminRole = @($roles | Where-Object { (Get-PropertyValue -Object $_ -Name "name") -eq "System Administrator" } | Select-Object -First 1)
$systemCustomizerRole = @($roles | Where-Object { (Get-PropertyValue -Object $_ -Name "name") -eq "System Customizer" } | Select-Object -First 1)
$systemAdminPrivilegeMap = @{}
$systemCustomizerPrivilegeMap = @{}

if ($systemAdminRole.Count -gt 0) {
    $systemAdminRoleId = ([string](Get-PropertyValue -Object $systemAdminRole[0] -Name "roleid")).ToLowerInvariant()
    if ($rolePrivilegeMapByRoleId.ContainsKey($systemAdminRoleId)) {
        $systemAdminPrivilegeMap = $rolePrivilegeMapByRoleId[$systemAdminRoleId]
    }
}
else {
    Write-Warning "System Administrator role was not found. SystemAdminSimilarityPercent will be blank."
}

if ($systemCustomizerRole.Count -gt 0) {
    $systemCustomizerRoleId = ([string](Get-PropertyValue -Object $systemCustomizerRole[0] -Name "roleid")).ToLowerInvariant()
    if ($rolePrivilegeMapByRoleId.ContainsKey($systemCustomizerRoleId)) {
        $systemCustomizerPrivilegeMap = $rolePrivilegeMapByRoleId[$systemCustomizerRoleId]
    }
}
else {
    Write-Warning "System Customizer role was not found. SystemCustomizerSimilarityPercent will be blank."
}

$results = foreach ($role in $roles) {
    $roleName = Get-PropertyValue -Object $role -Name "name"
    $roleTemplateId = [string](Get-PropertyValue -Object $role -Name "_roletemplateid_value")
    $existingRoleDescription = [string](Get-PropertyValue -Object $role -Name "description")
    $solutionId = [string](Get-PropertyValue -Object $role -Name "solutionid")
    $solution = if ($solutionById.ContainsKey($solutionId)) { $solutionById[$solutionId] } else { $null }
    $publisher = if ($solution) { Get-PropertyValue -Object $solution -Name "publisherid" } else { $null }

    $publisherName = [string](Get-PropertyValue -Object $publisher -Name "friendlyname")
    $publisherUniqueName = [string](Get-PropertyValue -Object $publisher -Name "uniquename")
    $publisherPrefix = [string](Get-PropertyValue -Object $publisher -Name "customizationprefix")
    $isMicrosoftPublisher = Test-MicrosoftPublisher -PublisherName $publisherName -PublisherUniqueName $publisherUniqueName -CustomizationPrefix $publisherPrefix
    $solutionName = if ($solution) { Get-PropertyValue -Object $solution -Name "friendlyname" } else { $null }
    $solutionUniqueName = if ($solution) { Get-PropertyValue -Object $solution -Name "uniquename" } else { $null }
    $hasRoleTemplate = -not [string]::IsNullOrWhiteSpace($roleTemplateId)
    $isManaged = [bool](Get-PropertyValue -Object $role -Name "ismanaged")
    $isKnownPredefinedRole = Test-KnownPredefinedRole -RoleName $roleName
    $productName = Get-MicrosoftProductName -RoleName $roleName -IsKnownPredefinedRole $isKnownPredefinedRole -SolutionName $solutionName -SolutionUniqueName $solutionUniqueName -PublisherPrefix $publisherPrefix
    $productRolePurpose = Get-ProductRolePurpose -RoleName $roleName -ProductName $productName -IsKnownPredefinedRole $isKnownPredefinedRole
    $roleDocumentation = Get-KnownRoleDocumentation -RoleName $roleName -ProductName $productName -ProductRolePurpose $productRolePurpose -IsKnownPredefinedRole $isKnownPredefinedRole -ExistingRoleDescription $existingRoleDescription
    $isKnownD365ProductRole = -not [string]::IsNullOrWhiteSpace($productName)
    $origin = Get-RoleOrigin -IsKnownPredefinedRole $isKnownPredefinedRole -IsKnownD365ProductRole $isKnownD365ProductRole -ProductName $productName -HasRoleTemplate $hasRoleTemplate -IsManaged $isManaged -IsMicrosoftPublisher $isMicrosoftPublisher -PublisherName $publisherName

    # compute privilege map for this role and evaluate admin risk
    $roleId = [string](Get-PropertyValue -Object $role -Name "roleid")
    $roleIdLower = if ($roleId) { $roleId.ToLowerInvariant() } else { $null }
    $rolePrivilegeMap = if ($roleIdLower -and $rolePrivilegeMapByRoleId.ContainsKey($roleIdLower)) { $rolePrivilegeMapByRoleId[$roleIdLower] } else { @{} }
    $adminRisk = Get-AdminRiskProfile -RolePrivilegeMap $rolePrivilegeMap -SystemAdminPrivilegeMap $systemAdminPrivilegeMap -SystemCustomizerPrivilegeMap $systemCustomizerPrivilegeMap

    [pscustomobject]@{
        RoleName              = $roleName
        Category              = $origin.Category
        Evidence              = $origin.Evidence
        IsKnownPredefinedRole = $isKnownPredefinedRole
        IsKnownD365ProductRole = $isKnownD365ProductRole
        ProductName           = $productName
        ProductRolePurpose    = $productRolePurpose
        RoleDescription       = $roleDocumentation.Description
        AccessGranted         = $roleDocumentation.AccessGranted
        SourceUrl             = $roleDocumentation.SourceUrl
        AdminRiskLevel                    = if ($adminRisk) { $adminRisk.AdminRiskLevel } else { $null }
        SystemAdminSimilarityPercent      = if ($adminRisk) { $adminRisk.SystemAdminSimilarityPercent } else { $null }
        SystemCustomizerSimilarityPercent = if ($adminRisk) { $adminRisk.SystemCustomizerSimilarityPercent } else { $null }
        HighRiskPrivilegeCount            = if ($adminRisk) { $adminRisk.HighRiskPrivilegeCount } else { 0 }
        GlobalHighRiskPrivilegeCount      = if ($adminRisk) { $adminRisk.GlobalHighRiskPrivilegeCount } else { 0 }
        CriticalGlobalPrivilegeCount      = if ($adminRisk) { $adminRisk.CriticalGlobalPrivilegeCount } else { 0 }
        HighRiskPrivilegeNames            = if ($adminRisk) { $adminRisk.HighRiskPrivilegeNames } else { $null }
        AdminRiskEvidence                 = if ($adminRisk) { $adminRisk.AdminRiskEvidence } else { $null }
        IsManaged             = $isManaged
        RoleTemplateName      = if ($roleTemplateById.ContainsKey($roleTemplateId)) { $roleTemplateById[$roleTemplateId] } else { $null }
        RoleTemplateId        = $roleTemplateId
        SolutionName          = $solutionName
        SolutionUniqueName    = $solutionUniqueName
        PublisherName         = $publisherName
        PublisherUniqueName   = $publisherUniqueName
        PublisherPrefix       = $publisherPrefix
        CreatedBy             = Get-FormattedValue -Object $role -LookupName "_createdby_value"
        CreatedOn             = Get-PropertyValue -Object $role -Name "createdon"
        ModifiedBy            = Get-FormattedValue -Object $role -LookupName "_modifiedby_value"
        ModifiedOn            = Get-PropertyValue -Object $role -Name "modifiedon"
        BusinessUnit          = Get-FormattedValue -Object $role -LookupName "_businessunitid_value"
        RoleId                = Get-PropertyValue -Object $role -Name "roleid"
        RoleIdUnique          = Get-PropertyValue -Object $role -Name "roleidunique"
        ParentRootRoleId      = Get-PropertyValue -Object $role -Name "_parentrootroleid_value"
    }
}
